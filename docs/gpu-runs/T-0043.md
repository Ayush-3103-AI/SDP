2. Write and run the demo-data script (PowerShell, from the repo root; copy the whole block):
```powershell
mkdir -Force runs | Out-Null
@'
from pathlib import Path

import cv2
import numpy as np

from xqi.store import Store
from xqi.types import CauseEstimate, CauseSet, ConfirmedDefect, EventVerdict, GateDecision, LayerResult, Proposal

root = Path("runs/_study_demo")
RUNS = {  # run_id: (cause, defect class, [(cause, var, from, delta, unit-gcode)])
    "demo-flow": ("flow_low", "under_extrusion",
                  [("flow_low", "flow", 100.0, 8.0, "M221 S108"), ("speed_high", "speed", 100.0, -10.0, "M220 S90")]),
    "demo-hot": ("nozzle_temp_high", "stringing",
                 [("nozzle_temp_high", "nozzle_temp", 210.0, -5.0, "M104 S205"),
                  ("speed_low", "speed", 100.0, 10.0, "M220 S110")]),
}
for run_id, (cause, cls, props) in RUNS.items():
    store = Store(root / run_id / "events.sqlite")
    store.write_run({"run_id": run_id, "mode": "advisory", "cfg": {"conformal": {"alpha": 0.1}}})
    for i in range(14):
        d = f"{run_id}-{i:02d}"
        img = np.full((160, 160, 3), 70, np.uint8)
        cv2.rectangle(img, (30, 30), (130, 130), (0, 0, 255), 2)
        cv2.putText(img, f"{cls} {i}", (8, 20), cv2.FONT_HERSHEY_SIMPLEX, 0.45, (255, 255, 255), 1)
        crop = root / run_id / "crops" / f"{d}.png"
        crop.parent.mkdir(parents=True, exist_ok=True)
        cv2.imwrite(str(crop), img)
        store.write_defect(ConfirmedDefect(d, cls, i, float(i), i + 0.5, 0.8, (30, 30, 130, 130), 0.05, 3, 0.4, False),
                           crop_path=str(crop))
        right = i % 4 != 3  # every 4th card the model ranks the wrong cause first
        p = 0.65 if right else 0.35
        probs = {props[0][0]: p, props[1][0]: 1 - p}
        top = max(probs, key=probs.get)
        store.write_explanation(d, CauseEstimate(probs, top, probs[top]), CauseSet(tuple(probs), 0.1, 0.6), 0.6, [],
                                i + 0.6)
        ids = []
        for c, var, x, delta, gcode in props:
            pid = f"{d}-{c}"
            ids.append(pid)
            layers = (LayerResult("L2", True, f"L2: {delta:+g} within limit"),)
            store.write_proposal(Proposal(pid, d, c, var, x, delta, x + delta, (gcode,),
                                          f"If {var} goes {x:g} -> {x + delta:g}, the {cls} should shrink."),
                                 GateDecision(pid, True, False, delta, x + delta, (gcode,), layers))
        store.write_verdict(EventVerdict(d, "PROPOSE", "two causes in the set", tuple(ids)), t_gate=i + 0.7)
    store.close()
(root / "campaign.csv").write_text("run_no,cause,mode,run_id\n1,flow_low,advisory,demo-flow\n"
                                   "2,nozzle_temp_high,advisory,demo-hot\n", encoding="utf-8")
print("ok", root)
'@ | Set-Content -Encoding utf8 runs\t0043_demo.py
uv run python runs\t0043_demo.py
```

## If it fails
- `0 events with a known cause; the study needs 24` on Start: step 2 did not run or `--events` /
  `--campaign` point elsewhere. Re-run step 2 and pass both paths exactly as in step 3.
- `No such file: config/causes.yaml`: start step 3 from the repo root.
- No picture on the cards: step 2's `runs/_study_demo/<run>/crops/` is missing; re-run step 2 from the repo root.
- Port 8501 busy: `uv run streamlit run xqi/study.py --server.port 8502 -- --events runs/_study_demo --campaign runs/_study_demo/campaign.csv --out reports/study/pilot.sqlite`
- A click seems to do nothing on **Stop**: tick "Confirm Stop (M112)" first (same as the dashboard).
- Restarting mid-pilot: delete `reports/study/pilot.sqlite`, refresh the browser and start again with ID 1.
